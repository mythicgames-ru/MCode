def estimate_message_tokens(messages):
    if not messages:
        return 0
    total = 3
    for m in messages:
        if not isinstance(m, dict):
            continue
        total += 4
        role = m.get("role")
        if role: total += _enc_len(role)
        name = m.get("name")
        if name: total += _enc_len(name)
        content = m.get("content")
        if isinstance(content, list):
            for part in content:
                if isinstance(part, dict):
                    total += _enc_len(part.get("text") or "")
                else:
                    total += _enc_len(part)
        else:
            total += _enc_len(content)
        tc = m.get("tool_calls")
        if tc:
            try:
                total += _enc_len(json.dumps(tc, ensure_ascii=False))
            except Exception:
                pass
    return max(1, total)

def normalize_messages(messages):
    """
    Приводит все сообщения к каноничному виду, чтобы хэш префикса
    не менялся между ходами из-за разного формата content (str vs list).
    Модель от этого не страдает — Anthropic/OpenAI принимают оба формата.
    """
    out = []
    for m in messages or []:
        if not isinstance(m, dict):
            continue
        nm = {"role": m.get("role") or "user"}

        # опциональные поля, которые могут влиять на префикс
        if m.get("name") is not None:
            nm["name"] = m["name"]
        if m.get("tool_call_id") is not None:
            nm["tool_call_id"] = m["tool_call_id"]
        if m.get("tool_calls") is not None:
            nm["tool_calls"] = m["tool_calls"]

        content = m.get("content")
        # каноничная форма: список блоков
        if isinstance(content, str):
            nm["content"] = [{"type": "text", "text": content}]
        elif isinstance(content, list):
            blocks = []
            for part in content:
                if isinstance(part, str):
                    blocks.append({"type": "text", "text": part})
                elif isinstance(part, dict):
                    # срезаем cache_control если уже был — добавим сами
                    clean = {k: v for k, v in part.items() if k != "cache_control"}
                    blocks.append(clean)
                else:
                    blocks.append({"type": "text", "text": str(part)})
            nm["content"] = blocks
        elif content is None:
            # допускаем пустой content (бывает у assistant с tool_calls)
            nm["content"] = []
        else:
            nm["content"] = [{"type": "text", "text": str(content)}]

        out.append(nm)
    return out


def _msg_hash(m):
    """Короткий хэш сообщения для отладки кэша."""
    try:
        s = json.dumps(m, sort_keys=True, ensure_ascii=False)
    except Exception:
        s = str(m)
    return hashlib.sha1(s.encode("utf-8")).hexdigest()[:8]


def _apply_cache_block(m, ttl):
    """Ставит cache_control на последний текстовый блок сообщения."""
    content = m.get("content") or []
    if not isinstance(content, list) or not content:
        # если блоков нет — добавим пустой
        m["content"] = [{"type": "text", "text": "", "cache_control": {"type": "ephemeral", "ttl": ttl}}]
        return m
    for block in reversed(content):
        if isinstance(block, dict) and block.get("type") == "text":
            block["cache_control"] = {"type": "ephemeral", "ttl": ttl}
            return m
    # если текстовых блоков нет — добавим пустой с меткой
    content.append({"type": "text", "text": "", "cache_control": {"type": "ephemeral", "ttl": ttl}})
    return m


def apply_anthropic_cache_control(messages, tools):
    """
    Расставляет ДО 4 брейкпоинтов Anthropic (лимит провайдера).

    Порядок важен — Anthropic проверяет префиксы от самого длинного:
      1) system            (стабилен весь диалог)
      2) tools             (стабилен весь диалог)
      3) последнее сообщение  — точка записи текущего хода
      4) предпоследнее сообщение — точка чтения следующего хода

    На следующем ходу предпоследнее сообщение станет СРЕДИНОЙ истории,
    и по нему (через скользящий сдвиг) прочитается весь предыдущий префикс.
    """
    if not CACHE_ENABLED:
        return messages, tools

    msgs = copy.deepcopy(messages)

    # 1) system
    for m in msgs:
        if m.get("role") == "system":
            _apply_cache_block(m, CACHE_TTL)
            break

    # 2) tools — breakpoint на последнем определении
    tools_out = copy.deepcopy(tools) if tools else []
    if tools_out:
        tools_out[-1] = dict(tools_out[-1])
        tools_out[-1]["cache_control"] = {"type": "ephemeral", "ttl": CACHE_TTL}

    # 3+4) последние два НЕ-системных сообщения
    non_sys_idx = [i for i, m in enumerate(msgs) if m.get("role") != "system"]
    if non_sys_idx:
        # последнее
        _apply_cache_block(msgs[non_sys_idx[-1]], CACHE_TTL)
        # предпоследнее (если есть)
        if len(non_sys_idx) >= 2:
            _apply_cache_block(msgs[non_sys_idx[-2]], CACHE_TTL)

    if CACHE_DEBUG:
        for i, m in enumerate(msgs):
            has_cc = any(
                isinstance(b, dict) and "cache_control" in b
                for b in (m.get("content") or [])
            )
            log(f"  msg[{i}] role={m.get('role')} hash={_msg_hash(m)} cc={'Y' if has_cc else 'N'}", "CACHE")

    return msgs, tools_out

def enc_body(p):
    n = secrets.token_bytes(12)
    c = AESGCM(SECRET).encrypt(n, p.encode("utf-8"), None)
    return base64.b64encode(n).decode() + "." + base64.b64encode(c).decode()


def dec_body(b):
    a = b.split(".", 1)
    if len(a) != 2: raise ValueError("bad enc")
    return AESGCM(SECRET).decrypt(base64.b64decode(a[0]), base64.b64decode(a[1]), None).decode("utf-8")


def _load_store():
    if not os.path.exists(USERS_FILE): return {"users": {}, "ip_reg": {}}
    try:
        with open(USERS_FILE, "r", encoding="utf-8") as f: raw = f.read().strip()
        if not raw: return {"users": {}, "ip_reg": {}}
        d = json.loads(dec_body(raw))
        if "users" not in d: d = {"users": d, "ip_reg": {}}
        return d
    except Exception as e:
        log(f"STORE load error: {e}", "ERR"); return {"users": {}, "ip_reg": {}}


def _save_store(s):
    t = USERS_FILE + ".tmp"
    with open(t, "w", encoding="utf-8") as f: f.write(enc_body(json.dumps(s, ensure_ascii=False)))
    os.replace(t, USERS_FILE)


def ensure_hour_reset(u):
    hb = hour_bucket()
    if u.get("hour_reset") != hb:
        u["tokens_used"] = 0
        u["hour_reset"] = hb


def ensure_day_reset(u):
    d = day_bucket()
    if u.get("search_day") != d:
        u["search_used"] = 0
        u["search_day"] = d


def find_user(users, token):
    if not token: return None, None
    for k, u in users.items():
        t = u.get("token") or ""
        try:
            if len(t) == len(token) and secrets.compare_digest(t, token): return k, u
        except Exception: pass
    return None, None


def client_ip():
    fwd = request.headers.get("X-Forwarded-For", "")
    if fwd:
        return fwd.split(",")[0].strip()
    return request.remote_addr or "unknown"


def cors(r):
    r.headers["Access-Control-Allow-Origin"] = "*"
    r.headers["Access-Control-Allow-Headers"] = "*"
    r.headers["Access-Control-Allow-Methods"] = "*"
    return r

@app.route("/api/search", methods=["POST", "OPTIONS"])
def search():
    if request.method == "OPTIONS": return cors(Response("")), 200
    t0 = time.time()

    try: data = json.loads(dec_body(request.data.decode("utf-8")))
    except Exception as e:
        log(f"SEARCH decrypt fail: {e}", "ERR")
        return cors(Response("bad request", status=400))

    token = data.get("token", "")
    query = (data.get("query") or "").strip()
    try:
        max_results = int(data.get("max_results") or 8)
    except Exception:
        max_results = 8
    max_results = max(1, min(20, max_results))

    if not query:
        return cors(Response(enc_body(json.dumps({"error": "empty query"})), content_type="text/plain"))

    with _lock:
        store = _load_store(); users = store["users"]
        _, u = find_user(users, token)
        if u is None:
            log("SEARCH invalid token", "ERR")
            return cors(Response(enc_body(json.dumps({"error": "invalid token"})), content_type="text/plain"))
        ensure_day_reset(u)
        u["search_used"] = int(u.get("search_used", 0)) + 1
        _save_store(store)

    log(f"SEARCH q='{query[:80]}' max={max_results}", "SEARCH")

    api_key = pick_tavily_key()
    if not api_key:
        return cors(Response(enc_body(json.dumps({"error": "no tavily keys configured"})), content_type="text/plain"))

    payload = {
        "api_key": api_key,
        "query": query,
        "max_results": max_results,
        "search_depth": "basic",
        "include_answer": True,
        "include_raw_content": False,
        "include_images": False,
    }

    try:
        r = requests.post(
            TAVILY_URL, json=payload,
            timeout=(TAVILY_TIMEOUT_CONNECT, TAVILY_TIMEOUT_READ),
        )
    except requests.exceptions.ConnectTimeout:
        log("TAVILY connect timeout", "ERR")
        return cors(Response(enc_body(json.dumps({"error": "tavily connect timeout"})), content_type="text/plain"))
    except requests.exceptions.ReadTimeout:
        log("TAVILY read timeout", "ERR")
        return cors(Response(enc_body(json.dumps({"error": "tavily read timeout"})), content_type="text/plain"))
    except Exception as e:
        log(f"TAVILY request fail: {e}", "ERR")
        return cors(Response(enc_body(json.dumps({"error": f"tavily request fail: {e}"})), content_type="text/plain"))

    if r.status_code != 200:
        detail = r.text[:400]
        log(f"TAVILY HTTP {r.status_code}: {detail}", "ERR")
        if r.status_code in (429, 432) and len(TAVILY_API_KEYS) > 1:
            backup = pick_tavily_key()
            if backup and backup != api_key:
                log("TAVILY retry with backup key", "SEARCH")
                payload["api_key"] = backup
                try:
                    r2 = requests.post(TAVILY_URL, json=payload,
                                       timeout=(TAVILY_TIMEOUT_CONNECT, TAVILY_TIMEOUT_READ))
                    if r2.status_code == 200:
                        r = r2
                    else:
                        return cors(Response(enc_body(json.dumps({
                            "error": f"tavily {r2.status_code}", "detail": r2.text[:400]
                        })), content_type="text/plain"))
                except Exception as e:
                    return cors(Response(enc_body(json.dumps({"error": f"tavily retry fail: {e}"})), content_type="text/plain"))
            else:
                return cors(Response(enc_body(json.dumps({
                    "error": f"tavily {r.status_code}", "detail": detail
                })), content_type="text/plain"))
        else:
            return cors(Response(enc_body(json.dumps({
                "error": f"tavily {r.status_code}", "detail": detail
            })), content_type="text/plain"))

    try:
        tj = r.json()
    except Exception as e:
        return cors(Response(enc_body(json.dumps({"error": f"tavily bad json: {e}"})), content_type="text/plain"))

    results = []
    for it in (tj.get("results") or [])[:max_results]:
        results.append({
            "title":   it.get("title") or "",
            "url":     it.get("url") or "",
            "content": it.get("content") or "",
            "score":   it.get("score") or 0,
        })

    out = {
        "query":   query,
        "answer":  tj.get("answer") or "",
        "results": results,
        "count":   len(results),
    }

    elapsed = int((time.time() - t0) * 1000)
    log(f"SEARCH done -> {len(results)} results in {elapsed}ms", "SEARCH")

    return cors(Response(enc_body(json.dumps(out, ensure_ascii=False)), content_type="text/plain"))

                              timeout=(UPSTREAM_CONNECT_TIMEOUT, UPSTREAM_READ_TIMEOUT),
                              headers=headers)
            log(f"← upstream connected in {time.time() - t_start:.2f}s status={r.status_code}", "UP")
            if r.status_code != 200:
                err = r.text[:600]
                yield _sse_line({"type": "error", "error": f"upstream {r.status_code}: {err}"})
                return

            try:
                for raw in r.iter_lines():
                    if not raw: continue
                    line = raw.decode("utf-8", errors="replace")
                    if not line.startswith("data: "): continue
                    piece = line[6:].strip()
                    if not piece: continue
                    if piece == "[DONE]":
                        got_done = True; break
                    try:
                        j = json.loads(piece)
                        u = j.get("usage")
                        if isinstance(u, dict):
                            pt = u.get("prompt_tokens")
                            ct = u.get("completion_tokens")
                            tt = u.get("total_tokens")
                            if pt is not None: last_prompt_tokens = int(pt)
                            if ct is not None: last_completion_tokens = int(ct)
                            if tt is not None: last_total_tokens = int(tt)
                            elif pt is not None and ct is not None:
                                last_total_tokens = last_prompt_tokens + last_completion_tokens

                            # OpenAI-style
                            ptd = u.get("prompt_tokens_details") or {}
                            itd = u.get("input_tokens_details") or {}
                            cached = ptd.get("cached_tokens") or itd.get("cached_tokens")
                            if cached is not None:
                                last_cached_tokens = int(cached)

                            # Anthropic-style
                            crt = u.get("cache_read_input_tokens")
                            if crt is not None:
                                last_cached_tokens = int(crt)
                            cct = u.get("cache_creation_input_tokens") or \
                                  u.get("cache_creation_tokens")
                            if cct is not None:
                                last_cache_creation_tokens = int(cct)
                            cc = u.get("cache_creation") or {}
                            if isinstance(cc, dict):
                                e5 = cc.get("ephemeral_5m_input_tokens")
                                e1 = cc.get("ephemeral_1h_input_tokens")
                                if e5 is not None or e1 is not None:
                                    last_cache_creation_tokens = (int(e5 or 0) + int(e1 or 0))
                    except Exception: pass
                    chunk_count += 1
                    yield _sse_line({"type": "raw", "chunk": piece})
            except requests.exceptions.ReadTimeout:
                log(f"UPSTREAM ReadTimeout: no data for {UPSTREAM_READ_TIMEOUT}s", "ERR")
                yield _sse_line({"type": "error",
                                 "error": f"upstream timeout: no data for {UPSTREAM_READ_TIMEOUT}s"})
                return
            except requests.exceptions.ConnectionError as e:
                log(f"UPSTREAM ConnectionError: {e}", "ERR")
                yield _sse_line({"type": "error", "error": f"upstream connection lost: {e}"})
                return

            if last_completion_tokens == 0 and last_total_tokens and last_prompt_tokens:
                last_completion_tokens = max(0, last_total_tokens - last_prompt_tokens)

            if COUNT_MODE == "total":
                charge = last_total_tokens
            elif COUNT_MODE == "completion":
                charge = last_completion_tokens
            else:
                ct = last_completion_tokens if last_completion_tokens > 0 else max(1, chunk_count * 2)
                charge = msg_tokens_est + ct

            charge = max(0, int(charge))

            cache_hit_ratio = 0.0
            if last_prompt_tokens > 0 and last_cached_tokens > 0:
                cache_hit_ratio = last_cached_tokens / last_prompt_tokens

            used_now = 0
            with _lock:
                store2 = _load_store()
                _, u2 = find_user(store2["users"], token)
                if u2 is not None:
                    ensure_hour_reset(u2)
                    u2["tokens_used"] = int(u2.get("tokens_used", 0)) + charge
                    used_now = u2["tokens_used"]
                    _save_store(store2)

            breakdown = (f"prompt={last_prompt_tokens} completion={last_completion_tokens} "
                         f"total_upstream={last_total_tokens} "
                         f"cached_read={last_cached_tokens} cache_new={last_cache_creation_tokens} "
                         f"hit={cache_hit_ratio*100:.1f}% "
                         f"msg_est={msg_tokens_est} charge={charge} mode={COUNT_MODE}")

            breakdown_dict = {
                "prompt": last_prompt_tokens,
                "completion": last_completion_tokens,
                "upstream_total": last_total_tokens,
                "cached_tokens": last_cached_tokens,
                "cache_creation_tokens": last_cache_creation_tokens,
                "cache_hit_ratio": round(cache_hit_ratio, 4),
                "messages_est": msg_tokens_est,
                "charged": charge,
                "mode": COUNT_MODE,
            }

            if not got_done:
                log(f"DONE(truncated) chunks={chunk_count} {breakdown} "
                    f"used={used_now}/{HOURLY_TOKEN_LIMIT} total={time.time() - t_start:.2f}s", "UP")
                yield _sse_line({"type": "done", "used": used_now, "limit": HOURLY_TOKEN_LIMIT,
                                 "tokens_this": charge, "truncated": True,
                                 "breakdown": breakdown_dict})
            else:
                log(f"DONE chunks={chunk_count} {breakdown} "
                    f"used={used_now}/{HOURLY_TOKEN_LIMIT} total={time.time() - t_start:.2f}s", "UP")
                yield _sse_line({"type": "done", "used": used_now, "limit": HOURLY_TOKEN_LIMIT,
                                 "tokens_this": charge,
                                 "breakdown": breakdown_dict})
        except requests.exceptions.ConnectTimeout:
            yield _sse_line({"type": "error", "error": "upstream connect timeout"})
        except Exception as e:
            log(f"CHAT exception: {e}", "ERR")
            yield _sse_line({"type": "error", "error": str(e)})

    return Response(gen(), content_type="text/event-stream",
                    headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no", "Connection": "keep-alive"})


def _sse_line(o): return "data: " + enc_body(json.dumps(o, ensure_ascii=False)) + "\n\n"


def _sse_error(m):
    def g(): yield _sse_line({"type": "error", "error": m})
    return Response(g(), content_type="text/event-stream")

def start_ngrok():
    try:
        from pyngrok import ngrok
        ngrok.set_auth_token(NGROK_AUTH_TOKEN)
        try:
            for t in ngrok.get_tunnels(): ngrok.disconnect(t.public_url)
        except Exception: pass
        try:
            tunnel = ngrok.connect(PORT, domain=NGROK_DOMAIN)
            log(f"NGROK: {tunnel.public_url} -> :{PORT}", "NGROK")
        except Exception as e:
            log(f"NGROK fixed-domain failed ({e}); trying random", "NGROK")
            tunnel = ngrok.connect(PORT)
            log(f"NGROK(rand): {tunnel.public_url}", "NGROK")
    except ImportError:
        log("pyngrok не установлен: pip install pyngrok", "NGROK")
    except Exception as e:
        log(f"ngrok fatal: {e}", "NGROK")
